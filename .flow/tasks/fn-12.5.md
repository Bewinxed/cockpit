# fn-12.5: Integrate dagre auto-layout

Add automatic hierarchical layout using dagre.

## Implementation

Create `apps/dashboard/src/lib/utils/flow-layout.ts`:

```typescript
import dagre from '@dagrejs/dagre';
import { Position, type Node, type Edge } from '@xyflow/svelte';

const NODE_WIDTH = 280;
const NODE_HEIGHT_MIN = 60;

export function layoutNodes(nodes: Node[], edges: Edge[]): Node[] {
  const g = new dagre.graphlib.Graph();
  g.setDefaultEdgeLabel(() => ({}));
  g.setGraph({ 
    rankdir: 'TB',  // Top to Bottom
    nodesep: 50,    // Horizontal spacing
    ranksep: 80,    // Vertical spacing
  });

  // Add nodes to dagre
  for (const node of nodes) {
    g.setNode(node.id, { 
      width: NODE_WIDTH, 
      height: node.data.height || NODE_HEIGHT_MIN 
    });
  }

  // Add edges to dagre
  for (const edge of edges) {
    g.setEdge(edge.source, edge.target);
  }

  // Run layout
  dagre.layout(g);

  // Apply positions
  return nodes.map(node => {
    const pos = g.node(node.id);
    return {
      ...node,
      position: {
        x: pos.x - NODE_WIDTH / 2,
        y: pos.y - (node.data.height || NODE_HEIGHT_MIN) / 2,
      },
      targetPosition: Position.Top,
      sourcePosition: Position.Bottom,
    };
  });
}
```
