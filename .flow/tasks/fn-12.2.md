# fn-12.2: Create FlowView container component

Create the main FlowView container that wraps svelte-flow with all controls.

## Component Structure

```svelte
<script>
  import { SvelteFlow, MiniMap, Controls, Background, Panel } from '@xyflow/svelte';
  import { nodeTypes } from './nodes';
  import '@xyflow/svelte/dist/style.css';
  
  let { instanceId, messages }: Props = $props();
  
  // Use $state.raw() for performance with svelte-flow
  let nodes = $state.raw<Node[]>([]);
  let edges = $state.raw<Edge[]>([]);
</script>

<div class="flow-container h-full w-full">
  <SvelteFlow 
    {nodes}
    {edges}
    {nodeTypes}
    fitView
    onlyRenderVisibleElements
  >
    <!-- Always visible MiniMap -->
    <MiniMap position="bottom-right" />
    
    <!-- Full control panel -->
    <Controls position="bottom-left" showZoom showFitView showLock />
    
    <!-- Background grid -->
    <Background />
    
    <!-- Custom panel for ActivityGrid + cost display -->
    <Panel position="top-left">
      <ActivityGrid {instanceId} />
      <CostDisplay {instanceId} />
    </Panel>
  </SvelteFlow>
</div>
```

## Features

- **MiniMap**: Always visible in bottom-right
- **Controls**: Full panel with zoom +/-, fit view, lock
- **Background**: Subtle grid pattern
- **Header panel**: ActivityGrid embedded + running cost total
- **Virtualization**: `onlyRenderVisibleElements` for performance
- **Auto-pan**: Follow new messages automatically

## Context Menu

Implement right-click context menu on nodes:
- Copy content
- Rewind to here
- Branch from here  
- Jump to chat view

```svelte
<script>
  let contextMenu = $state<{ x: number; y: number; node: Node } | null>(null);
  
  function onNodeContextMenu(event: CustomEvent) {
    const { event: e, node } = event.detail;
    e.preventDefault();
    contextMenu = { x: e.clientX, y: e.clientY, node };
  }
</script>

<SvelteFlow on:nodecontextmenu={onNodeContextMenu}>
  ...
</SvelteFlow>

{#if contextMenu}
  <ContextMenu 
    x={contextMenu.x} 
    y={contextMenu.y}
    node={contextMenu.node}
    onClose={() => contextMenu = null}
  />
{/if}
```
