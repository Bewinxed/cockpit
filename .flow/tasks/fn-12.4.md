# fn-12.4: Implement messages-to-nodes transformation

Create utility to transform Message[] into svelte-flow Node[] and Edge[].

## Approach

Leverage existing `groupedMessages` logic from WorkspaceInstance.svelte (lines 297-349) which already groups messages into:
- `'single'` - standalone user/assistant/system messages
- `'tool_group'` - consecutive tool_use + tool_result messages
- `'subagent_group'` - Task tool_use with parentToolUseId messages

## Data Flow

```typescript
messages: Message[] 
  → groupMessages(messages): MessageGroup[]
  → messagesToFlow(groups): { nodes: Node[], edges: Edge[] }
```

## Implementation

Create `apps/dashboard/src/lib/utils/flow-transform.ts`:

```typescript
import type { Node, Edge } from '@xyflow/svelte';
import type { Message, MessageGroup } from '$lib/stores/types';

interface FlowData {
  nodes: Node[];
  edges: Edge[];
}

export function messagesToFlow(
  groups: MessageGroup[],
  instanceId: string
): FlowData {
  const nodes: Node[] = [];
  const edges: Edge[] = [];
  
  let prevNodeId: string | null = null;
  
  for (const group of groups) {
    const nodeId = group.messages[0].id || group.messages[0].sdkUuid || crypto.randomUUID();
    
    // Determine node type
    const nodeType = getNodeType(group);
    
    nodes.push({
      id: nodeId,
      type: nodeType,
      position: { x: 0, y: 0 }, // dagre will position
      data: {
        group,
        messages: group.messages,
        instanceId,
      }
    });
    
    // Create edge from previous node
    if (prevNodeId) {
      edges.push({
        id: `e-${prevNodeId}-${nodeId}`,
        source: prevNodeId,
        target: nodeId,
      });
    }
    
    prevNodeId = nodeId;
  }
  
  return { nodes, edges };
}

function getNodeType(group: MessageGroup): string {
  if (group.type === 'tool_group') return 'tool';
  if (group.type === 'subagent_group') return 'subagent';
  
  const firstMsg = group.messages[0];
  if (firstMsg.type === 'user') return 'user';
  if (firstMsg.type === 'assistant') return 'assistant';
  return 'system';
}
```

## Subagent Handling

Messages with `parentToolUseId` belong to a subagent. These should:
1. Be excluded from main flow
2. Be stored in the subagent node's data
3. Rendered as nested flow when expanded
