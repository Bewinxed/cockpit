## Description
# fn-12.6: Add flow/chat view toggle to WorkspaceInstance

Add UI to switch between flow graph and traditional chat view with slide transition.

## UI Design

### Toggle Button (in InstanceHeader)
```svelte
<div class="flex items-center gap-1 border rounded-md p-0.5">
  <Button 
    variant={viewMode === 'chat' ? 'secondary' : 'ghost'} 
    size="icon-sm"
    onclick={() => setViewMode('chat')}
  >
    <MessageSquare class="w-4 h-4" />
  </Button>
  <Button 
    variant={viewMode === 'flow' ? 'secondary' : 'ghost'} 
    size="icon-sm"
    onclick={() => setViewMode('flow')}
  >
    <GitBranch class="w-4 h-4" />
  </Button>
</div>
```

### Keyboard Shortcut
- **Ctrl/Cmd + G** toggles between views
- Register in global keyboard handler

### Transition
- Slide animation between views (slide left/right)
- Preserve scroll position per view

## State Management

Store view preference per-instance in `ui.svelte.ts`:

```typescript
class UIStore {
  #instanceViewMode = new SvelteMap<string, 'chat' | 'flow'>();

  getViewMode(instanceId: string): 'chat' | 'flow' {
    return this.#instanceViewMode.get(instanceId) ?? 'flow';
  }
  
  setViewMode(instanceId: string, mode: 'chat' | 'flow') {
    this.#instanceViewMode.set(instanceId, mode);
  }

  getFlowState(instanceId: string) { ... }
  setFlowState(instanceId: string, state) { ... }
```

## WorkspaceInstance Changes

```svelte
<script>
  import { FlowView } from '$lib/components/features/flow';
  import { ui } from '$lib/stores';
  import { slide } from 'svelte/transition';
  
  const viewMode = $derived(ui.getViewMode(instanceId));
  
  function handleKeydown(e: KeyboardEvent) {
    if ((e.ctrlKey || e.metaKey) && e.key === 'g') {
      e.preventDefault();
      ui.setViewMode(instanceId, viewMode === 'flow' ? 'chat' : 'flow');
    }
  }
</script>

<svelte:window onkeydown={handleKeydown} />

{#if viewMode === 'flow'}
  <div transition:slide={{ axis: 'x' }}>
    <FlowView {instanceId} {messages} />
  </div>
{:else}
  <div transition:slide={{ axis: 'x' }}>
    <!-- Existing chat implementation -->
    <div class="chat-messages">
      {#each groupedMessages as group}
        ...
      {/each}
    </div>
  </div>
{/if}
```
